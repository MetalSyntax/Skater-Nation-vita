// Esta función mide el texto para los menús. En Android el juego usaba Paint.measureText()
// pero en Vita no tenemos la JVM corriendo, así que simulamos el ancho consultando
// la tabla de fuentes cargada en memoria. Si el puntero es NULL retorna 0 para no crashear.
int measure_text(const char *str, int font_id) {

    return get_font_width(font_id, str);
}

/**

 * @param[in] font_id Identificador de la fuente cargada en memoria.
